# `nautilus::region` Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add `nautilus::region` — inline, memoized isolated tracing scopes that record into the enclosing function's single `ExecutionTrace`.

**Architecture:** Unified root-region model (a function trace *is* its root region). Each `region(...)` creates a `RegionTraceContext` that hashes the parent env via hash-combine (`P`), records into the shared trace, and memoizes *closed* (non-escaping) executions keyed by `P`. On a memo hit the lambda is **not executed** (replay): in a RECORD-mode parent the next op merges into the recorded continuation; in a FOLLOW-mode parent the shared cursor jumps to the continuation.

**Tech Stack:** C++20, thread-local `TracingInterface` (ExceptionBased/Lazy contexts), `ExecutionTrace`, Catch2.

**Spec:** `docs/design/region.md` — the plan argues from the spec; executors read both.

#include "nautilus/Engine.hpp"
#include "nautilus/region.hpp"
#include "../common/ExecutionTest.hpp"
#include <catch2/catch_all.hpp>

using namespace nautilus;

TEST_CASE("region basic", "[region]") {
	nautilus::testing::forEachBackendWithTraceMode([](auto& engine) {
		auto fn = engine.template Compile<int64_t()>([]() {
			val<int64_t> sum = 0;
			region([&]() { sum = sum + 42; });
			return sum;
		});
		REQUIRE(fn() == 42);
	});
}

TEST_CASE("region preserves static unrolling", "[region]") {
	nautilus::testing::forEachBackendWithTraceMode([](auto& engine) {
		auto fn = engine.template Compile<int64_t()>([]() {
			val<int64_t> sum = 0;
			for (static_val<int32_t> j = 0; j < 3; j++) {
				region([&]() { sum = sum + val<int32_t>(j); });
			}
			return sum;
		});
		REQUIRE(fn() == 3); // 0+1+2; folding to 0+0+0 would give 0
	});
}

TEST_CASE("region replay skips closed body", "[region]") {
	nautilus::testing::forEachBackendWithTraceMode([](auto& engine) {
		auto fn = engine.template Compile<int64_t(int64_t)>([](val<int64_t> n) {
			val<int64_t> sum = 0;
			val<int64_t> i = 0;
			while (i < n) {
				region([&]() { sum = sum + 1; });
				i = i + 1;
			}
			return sum;
		});
		REQUIRE(fn(4) == 4);
		REQUIRE(fn(0) == 0);
	});
}

TEST_CASE("region escaped value lifecycle", "[region]") {
	nautilus::testing::forEachBackendWithTraceMode([](auto& engine) {
		auto fn = engine.template Compile<int64_t()>([]() {
			val<int64_t> out = 0;
			region([&]() { val<int64_t> t = 7; out = t; });
			return out + 1;
		});
		REQUIRE(fn() == 8);
	});
}

TEST_CASE("region internal branch", "[region]") {
	nautilus::testing::forEachBackendWithTraceMode([](auto& engine) {
		auto fn = engine.template Compile<int64_t(int64_t)>([](val<int64_t> x) {
			val<int64_t> sum = 0;
			region([&]() { if (x > 0) { sum = sum + 10; } else { sum = sum - 1; } });
			return sum;
		});
		REQUIRE(fn(1) == 10);
		REQUIRE(fn(-1) == -1);
	});
}

TEST_CASE("region nesting", "[region]") {
	nautilus::testing::forEachBackendWithTraceMode([](auto& engine) {
		auto fn = engine.template Compile<int64_t()>([]() {
			val<int64_t> sum = 0;
			region([&]() {
				region([&]() { sum = sum + 1; });
				sum = sum + 2;
			});
			return sum;
		});
		REQUIRE(fn() == 3);
	});
}

TEST_CASE("region empty and unnamed", "[region]") {
	nautilus::testing::forEachBackendWithTraceMode([](auto& engine) {
		auto fn = engine.template Compile<int64_t()>([]() {
			val<int64_t> sum = 0;
			region("empty", [&]() {});
			region([&]() {});
			return sum + 1;
		});
		REQUIRE(fn() == 1);
	});
}
