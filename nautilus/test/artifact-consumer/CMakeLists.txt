cmake_minimum_required(VERSION 3.22)
project(artifact_consumer LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(MLIR CONFIG QUIET)
find_package(nautilus CONFIG REQUIRED)
add_executable(artifact-consumer main.cpp)
target_link_libraries(artifact-consumer PRIVATE nautilus::nautilus)
if (CMAKE_SYSTEM_NAME STREQUAL "Linux")
    target_compile_options(artifact-consumer PRIVATE -fPIE)
    target_link_options(artifact-consumer PRIVATE -pie "LINKER:--build-id=sha1")
endif ()
