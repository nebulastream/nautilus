cmake_minimum_required(VERSION 3.22.1)
project(runtime_binding_consumer LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(NAUTILUS_CACHE_EXPECTED "Require the installed cache target" OFF)
find_package(MLIR CONFIG QUIET)
find_package(nautilus CONFIG REQUIRED)
add_executable(runtime-binding-consumer main.cpp)
target_link_libraries(runtime-binding-consumer PRIVATE nautilus::nautilus)
if (NAUTILUS_CACHE_EXPECTED)
    if (NOT TARGET nautilus::nautilus-cache)
        message(FATAL_ERROR "The installed package lacks nautilus::nautilus-cache")
    endif ()
    target_link_libraries(runtime-binding-consumer PRIVATE nautilus::nautilus-cache)
    target_compile_definitions(runtime-binding-consumer PRIVATE NAUTILUS_CACHE_EXPECTED)
elseif (TARGET nautilus::nautilus-cache)
    message(FATAL_ERROR "A cache-OFF package unexpectedly exports nautilus::nautilus-cache")
endif ()
if (CMAKE_SYSTEM_NAME STREQUAL "Linux")
    target_compile_options(runtime-binding-consumer PRIVATE -fPIE)
    target_link_options(runtime-binding-consumer PRIVATE -pie "LINKER:--build-id=sha1")
endif ()

include(CheckCXXSourceCompiles)
set(CMAKE_REQUIRED_LIBRARIES nautilus::nautilus)
foreach (feature ENABLE_COMPILER ENABLE_TRACING ENABLE_MLIR_BACKEND ENABLE_C_BACKEND
        ENABLE_BC_BACKEND ENABLE_TBC_BACKEND ENABLE_TBC_JIT ENABLE_ASMJIT_BACKEND)
    check_cxx_source_compiles("#include <nautilus/config.hpp>
#ifndef ${feature}
#error ${feature} is not configured
#endif
int main() { return 0; }" RUNTIME_BINDING_${feature})
endforeach ()
unset(CMAKE_REQUIRED_LIBRARIES)

enable_testing()
add_test(NAME runtime-binding-ordinary-interpreter COMMAND runtime-binding-consumer ordinary interpreter)
if (RUNTIME_BINDING_ENABLE_COMPILER AND RUNTIME_BINDING_ENABLE_TRACING)
    foreach (pair "ENABLE_MLIR_BACKEND,mlir" "ENABLE_C_BACKEND,cpp" "ENABLE_BC_BACKEND,bc"
            "ENABLE_TBC_BACKEND,tbc" "ENABLE_ASMJIT_BACKEND,asmjit" "ENABLE_TBC_JIT,tbc-auto")
        string(REPLACE "," ";" fields "${pair}")
        list(GET fields 0 feature)
        list(GET fields 1 backend)
        if (RUNTIME_BINDING_${feature})
            add_test(NAME runtime-binding-ordinary-${backend} COMMAND runtime-binding-consumer ordinary ${backend})
        endif ()
    endforeach ()
else ()
    add_test(NAME runtime-binding-ordinary-reduced COMMAND runtime-binding-consumer ordinary reduced)
endif ()

if (RUNTIME_BINDING_ENABLE_COMPILER AND RUNTIME_BINDING_ENABLE_TRACING AND RUNTIME_BINDING_ENABLE_MLIR_BACKEND
        AND CMAKE_SYSTEM_NAME STREQUAL "Linux" AND CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|amd64|AMD64)$")
    set(binding_artifact "${CMAKE_CURRENT_BINARY_DIR}/artifacts/bindings.artifact")
    add_test(NAME runtime-binding-artifact-emit COMMAND runtime-binding-consumer emit "${binding_artifact}")
    set_tests_properties(runtime-binding-artifact-emit PROPERTIES FIXTURES_SETUP runtime-binding-artifact)
    foreach (mode native bytecode rebound schema-mismatch native-no-bytecode native-corrupt-bytecode)
        add_test(NAME runtime-binding-artifact-${mode} COMMAND runtime-binding-consumer ${mode} "${binding_artifact}")
        set_tests_properties(runtime-binding-artifact-${mode} PROPERTIES FIXTURES_REQUIRED runtime-binding-artifact)
    endforeach ()
    if (NAUTILUS_CACHE_EXPECTED)
        set(binding_cache "${CMAKE_CURRENT_BINARY_DIR}/cache")
        add_test(NAME runtime-binding-cache-cold COMMAND runtime-binding-consumer cold "${binding_cache}")
        add_test(NAME runtime-binding-cache-warm COMMAND runtime-binding-consumer warm "${binding_cache}")
        add_test(NAME runtime-binding-cache-repair COMMAND runtime-binding-consumer repair "${binding_cache}")
        add_test(NAME runtime-binding-cache-warm-after-repair COMMAND runtime-binding-consumer warm "${binding_cache}")
        set_tests_properties(runtime-binding-cache-cold PROPERTIES FIXTURES_SETUP runtime-binding-cache-cold)
        set_tests_properties(runtime-binding-cache-warm PROPERTIES
                FIXTURES_REQUIRED runtime-binding-cache-cold FIXTURES_SETUP runtime-binding-cache-warm)
        set_tests_properties(runtime-binding-cache-repair PROPERTIES
                FIXTURES_REQUIRED runtime-binding-cache-warm FIXTURES_SETUP runtime-binding-cache-repaired)
        set_tests_properties(runtime-binding-cache-warm-after-repair PROPERTIES
                FIXTURES_REQUIRED runtime-binding-cache-repaired)
    endif ()
endif ()

get_property(binding_tests DIRECTORY PROPERTY TESTS)
set_tests_properties(${binding_tests} PROPERTIES TIMEOUT 180)
if (CMAKE_SYSTEM_NAME STREQUAL "Linux")
    set(binding_library_path "$<TARGET_FILE_DIR:nautilus::nautilus>")
    if (NAUTILUS_CACHE_EXPECTED)
        string(APPEND binding_library_path ":$<TARGET_FILE_DIR:nautilus::nautilus-cache>")
    endif ()
    set_tests_properties(${binding_tests} PROPERTIES
            ENVIRONMENT "LD_LIBRARY_PATH=${binding_library_path}:$ENV{LD_LIBRARY_PATH}")
endif ()
